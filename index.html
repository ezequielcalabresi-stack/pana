function renderizarPlanillaReparto() { const tbody = document.getElementById('tabla-reparto'); if (!tbody) return; tbody.innerHTML = ''; const pedidosHoy = dbRepartos[fechaActual] || []; let totalMonto = 0, totalEfectivo = 0, totalCtaCte = 0; pedidosHoy.forEach((p) => { totalMonto += p.monto; if (p.pagado && p.medioPago === 'Efectivo') totalEfectivo += p.monto; else if (!p.pagado) totalCtaCte += p.monto; // Se eliminó el filtro que ocultaba los automáticos. // Agregamos un indicador visual (🤖) para saber cuáles se cargaron solos const iconoAuto = p.origenAuto ? '🤖 ' : ''; const tr = document.createElement('tr'); tr.className = "border-b hover:bg-gray-50 text-xs md:text-sm"; tr.innerHTML = ` ${p.clienteNombre} ${iconoAuto}${p.producto} ${p.origen === 'despacho' ? `Despacho` : `Cuadra`} ${p.unidades} un. $${p.monto.toLocaleString()} ${p.pagado ? `Cobrado (${p.medioPago})` : `Pendiente`} ${!p.pagado ? `` : ``} `; tbody.appendChild(tr); }); document.getElementById('tot-rep-monto').textContent = `$${totalMonto.toLocaleString()}`; document.getElementById('tot-rep-efectivo').textContent = `$${totalEfectivo.toLocaleString()}`; document.getElementById('tot-rep-ctacte').textContent = `$${totalCtaCte.toLocaleString()}`; let totalTransf = 0; pedidosHoy.forEach(p => { if (p.pagado && p.medioPago !== 'Efectivo') totalTransf += p.monto; }); const elEf = document.getElementById('caja-rep-efectivo'); const elTr = document.getElementById('caja-rep-transf'); const elTot = document.getElementById('caja-rep-total'); if (elEf) elEf.textContent = `$${totalEfectivo.toLocaleString()}`; if (elTr) elTr.textContent = `$${totalTransf.toLocaleString()}`; if (elTot) elTot.textContent = `$${(totalEfectivo + totalTransf).toLocaleString()}`; }
