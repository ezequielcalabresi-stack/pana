function enviarResumenWhatsAppCliente() { if (!clienteSeleccionadoId) return; const cli = dbClientes[clienteSeleccionadoId]; if (!cli) return; // OBTENER VALORES DE LOS FILTROS DE FECHA const filtroDesde = document.getElementById('filtro-cta-desde')?.value || ''; const filtroHasta = document.getElementById('filtro-cta-hasta')?.value || ''; const movs = []; if (dbRepartos) { Object.entries(dbRepartos).forEach(([fecha, pDia]) => { if (!pDia) return; Object.values(pDia).forEach(p => { if (p && p.clienteId === clienteSeleccionadoId && !p.pagado) { movs.push({ fecha, concepto: `${p.producto} (${p.unidades} un.)`, monto: p.monto, tipo: '+' }); } }); }); } if (dbHistorialClientes && dbHistorialClientes[clienteSeleccionadoId]) { Object.values(dbHistorialClientes[clienteSeleccionadoId]).forEach(p => { if (p) { const esDebito = p.tipo === 'debito'; movs.push({ fecha: p.fecha, concepto: esDebito ? `Cargo manual: ${p.medio}` : `Pago (${p.medio})`, monto: p.monto, tipo: esDebito ? '+' : '-' }); } }); } movs.sort((a, b) => new Date(a.fecha || 0) - new Date(b.fecha || 0)); let acumuladoGeneral = 0; let movsFiltrados = []; let saldoFinalPeriodo = 0; // Primero calculamos el acumulado histórico paso a paso igual que la tabla movs.forEach(m => { const valorMonto = m.monto || 0; const impacto = m.tipo === '+' ? valorMonto : -valorMonto; acumuladoGeneral += impacto; // Si pasa el filtro de fecha, lo guardamos para el detalle y actualizamos el saldo del período const pasaDesde = !filtroDesde || m.fecha >= filtroDesde; const pasaHasta = !filtroHasta || m.fecha <= filtroHasta; if (pasaDesde && pasaHasta) { movsFiltrados.push(m); saldoFinalPeriodo = acumuladoGeneral; // Toma el saldo acumulado exacto de la última fila visible } }); // Si no hay filtro de fechas aplicado, el saldo final es el acumulado total general if (!filtroDesde && !filtroHasta) { saldoFinalPeriodo = acumuladoGeneral; } let periodoTexto = ""; if (filtroDesde && filtroHasta) { periodoTexto = ` (del ${filtroDesde} al ${filtroHasta})`; } else if (filtroDesde) { periodoTexto = ` (desde el ${filtroDesde})`; } else if (filtroHasta) { periodoTexto = ` (hasta el ${filtroHasta})`; } let mensaje = `Hola *${cli.nombre}*, te enviamos el resumen de tu Cuenta Corriente${periodoTexto}:\n\n`; if (movsFiltrados.length === 0) { mensaje += `No se registran movimientos en el período seleccionado.\n`; } else { mensaje += `*Detalle de movimientos del período:*\n`; movsFiltrados.forEach(m => { mensaje += `• ${m.fecha}: ${m.concepto} -> ${m.tipo}${(m.monto || 0).toLocaleString()}\n`; }); mensaje += `\n*SALDO TOTAL A PAGAR: $${saldoFinalPeriodo.toLocaleString()}*\n\n`; mensaje += `Muchas gracias por tu compra. La Casa del Pan.`; } let tel = (cli.telefono || '').replace(/\D/g, ''); const url = tel ? `https://api.whatsapp.com/send?phone=${tel}&text=${encodeURIComponent(mensaje)}` : `https://api.whatsapp.com/send?text=${encodeURIComponent(mensaje)}`; window.open(url, '_blank'); }
